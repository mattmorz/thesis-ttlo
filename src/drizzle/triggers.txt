CREATE OR REPLACE FUNCTION create_validation_record()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.requires_validation = true THEN
        INSERT INTO documents_validation (
            document_id,
            validation_status,
            created_at,
            updated_at
        ) VALUES (
            NEW.id,
            'pending',
            NOW(),
            NOW()
        );
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_create_validation
AFTER INSERT ON documents
FOR EACH ROW
EXECUTE FUNCTION create_validation_record();
